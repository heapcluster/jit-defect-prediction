5拷贝，保证「生成 → 静态目录」这一步不靠人工，且不覆盖已有文件（只在缺失或源更新时复制）。 

6"""

20def _copy_tree(src: str, dst) -> int: 

21 copied = 0 

22 for png in src.rglob("*.png"): 

23 rel = png.relative_to(src) 

24 target = dst / rel 

25 if target.is_file() and target.stat().st_mtime >= png.stat().st_mtime: 

26 continue 

27 target.parent.mkdir(parents=True, exist_ok=True) 

28 shutil.copy2(png, target) 

29 copied += 1 

30 return copied

33def sync_figures() -> int: 

34 """把 `data_model/pipelines/{line}/reports/figs_by_model` 同步到静态图目录。 

35 

36 返回本次复制的文件数；源目录缺失时跳过（接口侧再做降级）。 

37 """ 

38 copied = 0 

39 for line in _LINES: 

40 src = config.PIPELINES_DIR / line / "reports" / "figs_by_model" 

41 if not src.is_dir(): 

42 continue 

43 copied += _copy_tree(src, config.FIGURES_DIR / line) 

44 if copied: 

45 logger.info("figures synced: %s files -> %s", copied, config.FIGURES_DIR) 

46 return copied
