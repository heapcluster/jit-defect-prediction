95def _file_weight(path: str) -> tuple[float, str]: 

96 """启发式文件类型权重：源码 > 普通文件 > 配置/文档 > 测试/生成物。""" 

97 low = path.lower() 

98 if any(marker in low for marker in _GENERATED_MARKERS): 

99 return 0.1, "生成物" 

100 if any(marker in low for marker in _TEST_MARKERS): 

101 return 0.35, "测试文件" 

102 if low.endswith(_SOURCE_SUFFIXES): 102 ↛ 104line 102 didn't jump to line 104 because the condition on line 102 was always true

103 return 1.0, "源码文件" 

104 if low.endswith(_DOC_SUFFIXES): 

105 return 0.4, "文档/配置" 

106 return 0.6, "普通文件"

119def _file_raw_score(file: dict) -> tuple[float, list[str]]: 

120 added = len(file.get("added_lines") or []) 

121 deleted = int(file.get("deleted_count") or 0) 

122 hunks = file.get("hunks") or [] 

123 weight, kind = _file_weight(file["path"]) 

124 modified = bool(added and deleted) 

125 # 修改既有逻辑（有增有删）比纯新增文件更值得先看；hunk 越多说明改动越分散 

126 raw = (added + 2 * deleted) * (1.5 if modified else 1.0) * weight 

127 raw *= 1 + 0.15 * max(0, min(len(hunks) - 1, 4)) 

128 reasons = [kind] 

129 if modified: 

130 reasons.append("修改既有代码") 

131 if len(hunks) > 1: 131 ↛ 132line 131 didn't jump to line 132 because the condition on line 131 was never true

132 reasons.append(f"{len(hunks)} 个 hunk") 

133 return raw, reasons

136def _hunk_raw_score(hunk: dict) -> float: 

137 added = hunk.get("added_lines") or [] 

138 deleted = int(hunk.get("deleted_count") or 0) 

139 # 用“代码行权重”而不是纯行数：license/注释 hunk 不应压过真正的逻辑 hunk 

140 return ( 

141 sum(_line_content_weight(line) for line in added) 

142 + 2 * deleted 

143 + (1 if added and deleted else 0) 

144 )

177def _line_content_weight(line: dict) -> float: 

178 """单行内容信号：空行/注释/import 低，控制流与边界/安全关键词高。""" 

179 text = (line.get("content") or "").strip() 

180 if not text: 180 ↛ 181line 180 didn't jump to line 181 because the condition on line 180 was never true

181 return 0.1 

182 if text.startswith(("//", "/*", "*", "#", "<!--")): 

183 return 0.15 

184 if text.startswith(("import ", "package ", "from ", "using ")): 184 ↛ 185line 184 didn't jump to line 185 because the condition on line 184 was never true

185 return 0.3 

186 low = text.lower() 

187 return 1.5 if any(keyword in low for keyword in _SUSPICIOUS_KEYWORDS) else 1.0

190def _line_suspicion( 

191 line: dict, *, risk_score: float | None, file_score: float, hunk_norm: float, path: str 

192) -> float: 

193 """启发式逐行可疑度：提交风险 × 文件/hunk 权重 × 行内容信号（不是训练出的行级概率）。""" 

194 factor = _line_content_weight(line) 

195 _, kind = _file_weight(path) 

196 factor *= _LINE_PATH_FACTOR.get(kind, 0.9) 

197 base = risk_score if risk_score is not None else 0.5 

198 score = base * (0.45 + 0.35 * file_score + 0.20 * hunk_norm) * factor 

199 return round(max(0.02, min(score, 0.99)), 3)
