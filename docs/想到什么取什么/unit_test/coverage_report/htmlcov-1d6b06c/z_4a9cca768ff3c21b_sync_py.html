86def _unpredicted( 

87 hashes: list[str], models: set[str], limit: int | None = None 

88) -> tuple[list[str], bool]: 

89 """按输入顺序挑出缺「models 中任一模型」预测行的哈希；分批 IN，不整表载入。 

90 

91 `models` 为空集时无缺口（全部视为已覆盖）。`limit` 非空时最多返回 limit 条， 

92 并在「还有更多」时把第二项置为 True。 

93 """ 

94 wanted = set(models) 

95 if not wanted: 95 ↛ 96line 95 didn't jump to line 96 because the condition on line 95 was never true

96 return [], False 

97 pending: list[str] = [] 

98 with SessionLocal() as session: 

99 for start in range(0, len(hashes), _PREDICTED_CHUNK): 

100 chunk = hashes[start : start + _PREDICTED_CHUNK] 

101 rows = session.execute( 

102 select(Prediction.commit_hash, Prediction.model_name).where( 

103 Prediction.commit_hash.in_(chunk) 

104 ) 

105 ) 

106 have: dict[str, set[str]] = {} 

107 for commit_hash, model_name in rows: 

108 have.setdefault(commit_hash, set()).add(model_name) 

109 covered = {h for h, names in have.items() if wanted <= names} 

110 for commit_hash in chunk: 

111 if commit_hash in covered: 

112 continue 

113 if limit is not None and len(pending) >= limit: 

114 return pending, True 

115 pending.append(commit_hash) 

116 return pending, False

129def _prefetch_features(hashes: list[str], repo) -> int: 

130 """批量现算缺失的 14 维特征并落库（已有 commit_feature 的提交跳过）。 

131 

132 逐条预测各自调用 extract_single_commit 会各扫一遍全历史；这里一次扫描算完 

133 整批缺失特征，N 条新提交的耗时从 N×110s 降到约 1×110s + 批量打分。 

134 """ 

135 if not hashes: 

136 return 0 

137 with SessionLocal() as session: 

138 have = { 

139 commit_hash 

140 for (commit_hash,) in session.query(CommitFeature.commit_hash) 

141 .filter(CommitFeature.commit_hash.in_(hashes)) 

142 .all() 

143 } 

144 missing = [commit_hash for commit_hash in hashes if commit_hash not in have] 

145 if not missing: 

146 return 0 

147 try: 

148 items = feature_source.extract_many(missing, repo) 

149 return store_extracted_features(items) 

150 except Exception: # noqa: BLE001 —— 预取失败退回逐条现算，不阻断同步 

151 logger.exception("batch feature prefetch failed; fallback to per-commit extraction") 

152 return 0

169def _run_batch( 

170 *, 

171 mode: str, 

172 scope: str, 

173 models: set[str], 

174 plan: Callable[[str, str], tuple[int, list[str], bool | None]], 

175) -> dict: 

176 """取锁 → （可选 fetch）→ 按 plan 算待预测 → 批量抽特征 → 逐条预测入库 → 归还锁。""" 

177 if not _sync_lock.acquire(blocking=False): 

178 raise AppError(40001, SYNC_BUSY_MESSAGE) 

179 sync_state.set_state( 

180 running=True, 

181 phase="fetch" if config.SYNC_FETCH else "plan", 

182 mode=mode, 

183 scope=scope, 

184 total=0, 

185 done=0, 

186 current_hash=None, 

187 started_at=datetime.now(UTC).replace(tzinfo=None), 

188 last_error=None, 

189 ) 

190 try: 

191 repo = config.SOURCE_REPO 

192 if repo is None or not (repo / ".git").exists(): 

193 raise AppError(40400, "sync source repository unavailable") 

194 

195 tip = ( 

196 fetch_latest_tip(str(repo)) 

197 if config.SYNC_FETCH 

198 else run_git(str(repo), "rev-parse", "HEAD").strip() 

199 ) 

200 sync_state.set_state(phase="plan") 

201 scanned, pending, truncated = plan(str(repo), tip) 

202 sync_state.set_state(total=len(pending), phase="extract" if pending else "predict") 

203 

204 model_names = sorted(models) 

205 extract_repo, release = ensure_extractable(str(repo), tip) 

206 try: 

207 extracted = _prefetch_features(pending, extract_repo) 

208 if pending: 

209 logger.info("sync prefetch: pending=%s extracted=%s", len(pending), extracted) 

210 sync_state.set_state(phase="predict") 

211 results: list[dict] = [] 

212 skipped: list[dict] = [] 

213 for commit_hash in pending: 

214 sync_state.set_state(current_hash=commit_hash) 

215 try: 

216 _predict_commit_models(commit_hash, extract_repo, model_names) 

217 results.append({"commit_hash": commit_hash}) 

218 except AppError as exc: 

219 skipped.append( 

220 {"commit_hash": commit_hash, "code": exc.code, "message": exc.message} 

221 ) 

222 _update_done(results, skipped) 

223 finally: 

224 release() 

225 

226 logger.info( 

227 "sync done: mode=%s tip=%s scanned=%s pending=%s predicted=%s skipped=%s truncated=%s", 

228 mode, 

229 tip[:10], 

230 scanned, 

231 len(pending), 

232 len(results), 

233 len(skipped), 

234 truncated, 

235 ) 

236 data: dict = { 

237 "tip": tip, 

238 "scanned": scanned, 

239 "new": len(pending), 

240 "predicted": results, 

241 "skipped": skipped, 

242 } 

243 if truncated is not None: 

244 data["truncated"] = truncated 

245 return data 

246 except Exception as exc: 

247 sync_state.set_state(last_error=str(exc)[:200]) 

248 raise 

249 finally: 

250 sync_state.set_state(running=False, phase=None) 

251 _sync_lock.release()

320@router.post("/api/sync") 

321def sync_endpoint( 

322 line: str | None = Query(default=None), 

323 model_name: str | None = Query(default=None), 

324 start_time: str | None = Query(default=None), 

325 end_time: str | None = Query(default=None), 

326) -> dict: 

327 """按架构 / 模型 / commit 时间区间**异步**补算入库（docs/api/sync.md）。 

328 

329 提交后立即返回；进度轮 `/api/sync/status`。同一时刻只允许一个同步任务， 

330 已在同步中时返回 40001（可解释冲突，不排队）。 

331 """ 

332 if line is not None and line not in VALID_LINES: 

333 raise AppError(40001, "参数非法：line 取值需为 expert / semantic / fused / e2e") 

334 if model_name is not None and not model_registry.has(model_name): 

335 raise AppError(40001, "参数非法：model_name 不存在") 

336 start = parse_time_param(start_time, "start_time") if start_time else None 

337 end = parse_time_param(end_time, "end_time", "end") if end_time else None 

338 # 「检查 running + 置 running=True」必须在锁内原子完成：否则两个并发请求都能通过 

339 # 检查、各 spawn 一个 worker，败者 worker 的异常处理会把 running=False 写回， 

340 # 覆盖胜者正在跑的状态。 

341 with _sync_lock: 

342 if sync_state.get_sync_state()["running"]: 

343 raise AppError(40001, SYNC_BUSY_MESSAGE) 

344 # 在请求内先置 running=True：否则前端提交后立即轮询时，worker 可能还没启动， 

345 # 会读到上一次任务的 running=false 而误判「已完成」。 

346 sync_state.set_state( 

347 running=True, 

348 phase="fetch" if config.SYNC_FETCH else "plan", 

349 mode="manual", 

350 scope="latest", 

351 total=0, 

352 done=0, 

353 current_hash=None, 

354 started_at=datetime.now(UTC).replace(tzinfo=None), 

355 last_error=None, 

356 ) 

357 _spawn_sync_worker(lambda: sync_manual(line=line, model_name=model_name, start=start, end=end)) 

358 return ok({"accepted": True})
