24_GIT_TIMEOUT = 180 

25# 可重试的失败特征：只对「网络抖动」重试；仓库不存在 / 权限 / 参数错重试没有意义 

26_NETWORK_HINTS = ( 

27 "could not resolve host", 

28 "connection timed out", 

29 "connection reset", 

30 "failed to connect", 

31 "early eof", 

32 "rpc failed", 

33 "the remote end hung up", 

34 "ssl", 

35 "tls", 

36 "timeout", 

37) 

38# 日志里的 URL 可能含 `https://token@host`，脱敏后再落盘 

39_CREDENTIAL = re.compile(r"://[^/@\s]+@")

42class GitError(AppError): 

43 """git 子进程失败的统一异常（沿用契约错误码 50000）。 

44 

45 `retryable` 区分「网络抖动（值得重试）」与「配置/仓库错误（立即失败）」—— 

46 前者重试 3 次、每次 5s 等待，后者直接把失败抛给调用方。 

47 """ 

48 

49 def __init__(self, *, retryable: bool) -> None: 

50 super().__init__(50000, INTERNAL_ERROR_MESSAGE) 

51 self.retryable = retryable

54def _redact(text: str) -> str: 

55 """抹掉 URL 里的凭据片段，避免 token 落进日志。""" 

56 return _CREDENTIAL.sub("://***@", text)

59def _looks_like_network(stderr: str) -> bool: 

60 lowered = stderr.lower() 

61 return any(hint in lowered for hint in _NETWORK_HINTS)

64def run_git(repo: str, *args: str) -> str: 

65 """执行 git 命令；代理只认 `GIT_PROXY`（其次 `HTTPS_PROXY`）。 

66 

67 `encoding="utf-8"` 显式指定：Windows 默认按 locale（cp936）解码，Camel 的 

68 中文提交信息会乱码并最终写进 `commit.message`。 

69 """ 

70 cmd = ["git", "-C", repo] 

71 proxy = os.getenv("GIT_PROXY", "").strip() or os.getenv("HTTPS_PROXY", "").strip() 

72 if proxy: 72 ↛ 75line 72 didn't jump to line 75 because the condition on line 72 was always true

73 cmd += ["-c", f"http.proxy={proxy}", "-c", f"https.proxy={proxy}"] 

74 else: 

75 cmd += ["-c", "http.proxy=", "-c", "https.proxy="] 

76 cmd += [*args] 

77 try: 

78 proc = subprocess.run( 

79 cmd, 

80 capture_output=True, 

81 text=True, 

82 encoding="utf-8", 

83 errors="replace", 

84 timeout=_GIT_TIMEOUT, 

85 check=True, 

86 ) 

87 except subprocess.TimeoutExpired as exc: 

88 logger.warning("git 超时（%ss）：%s", _GIT_TIMEOUT, args[:2]) 

89 raise GitError(retryable=True) from exc 

90 except subprocess.CalledProcessError as exc: 

91 # 失败原因几乎全在 stderr 里；只记 args 会让线上「同步全挂」无从下手 

92 stderr = _redact((exc.stderr or "").strip()) 

93 logger.warning("git 失败（%s）：%s", args[:2], stderr[:500]) 

94 raise GitError(retryable=_looks_like_network(stderr)) from exc 

95 except OSError as exc: 

96 # git 未安装 / 不可执行：重试也没用 

97 logger.warning("git 无法执行（%s）：%s", args[:2], exc) 

98 raise GitError(retryable=False) from exc 

99 return proc.stdout

112def fetch_latest_tip(repo: str, retries: int = 3) -> str: 

113 """fetch 远端并返回最新 tip 哈希（FETCH_HEAD 首个条目）；仅网络类失败重试。""" 

114 last: Exception | None = None 

115 for attempt in range(1, retries + 1): 

116 try: 

117 run_git(repo, *build_fetch_args()) 

118 break 

119 except GitError as exc: 

120 if not exc.retryable: 

121 raise 

122 last = exc 

123 if attempt < retries: 

124 time.sleep(5) 

125 else: 

126 raise AppError(50000, INTERNAL_ERROR_MESSAGE) from last 

127 tip = run_git(repo, "rev-parse", "FETCH_HEAD").strip() 

128 if len(tip) != 40: 

129 raise AppError(50000, INTERNAL_ERROR_MESSAGE) 

130 return tip

133def is_ancestor(repo: str, commit: str, base: str = "HEAD") -> bool: 

134 """判断 commit 是否为 base 的祖先（纯函数，供测试）。""" 

135 check = subprocess.run( 

136 ["git", "-C", repo, "merge-base", "--is-ancestor", commit, base], 

137 capture_output=True, 

138 encoding="utf-8", 

139 errors="replace", 

140 timeout=_GIT_TIMEOUT, 

141 check=False, 

142 ) 

143 return check.returncode == 0

146def ensure_extractable(repo: str, tip: str) -> tuple[Path | str, Callable[[], None]]: 

147 """返回 (特征提取用仓库路径, 释放函数)。 

148 

149 tip 已从 HEAD 可达直接用 SOURCE_REPO；否则建临时 worktree，用后即焚。 

150 绝不改动 SOURCE_REPO 的检出状态。 

151 """ 

152 if is_ancestor(repo, tip): 

153 return Path(repo), lambda: None 

154 

155 tmp = Path(tempfile.mkdtemp(prefix="jit-sync-")) 

156 try: 

157 run_git(repo, "worktree", "add", "--detach", "--no-checkout", str(tmp), tip) 

158 except AppError: 

159 shutil.rmtree(tmp, ignore_errors=True) 

160 raise 

161 

162 def _release() -> None: 

163 try: 

164 run_git(repo, "worktree", "remove", "--force", str(tmp)) 

165 finally: 

166 shutil.rmtree(tmp, ignore_errors=True) 

167 

168 return str(tmp), _release

188def list_hashes_in_range(repo: str, start, end, ref: str = "FETCH_HEAD") -> list[str]: 

189 """取 ref 在 [start, end]（两端闭区间，commit date）内的哈希，时间倒序。 

190 

191 `start` / `end` 为 naive datetime（可能为 None，表示该侧不设界）。 

192 """ 

193 args = ["log", "--format=%H"] 

194 if start is not None: 

195 args += [f"--since={start.strftime('%Y-%m-%d %H:%M:%S')}"] 

196 if end is not None: 

197 args += [f"--until={end.strftime('%Y-%m-%d %H:%M:%S')}"] 

198 args.append(ref) 

199 out = run_git(repo, *args) 

200 return [line.strip() for line in out.splitlines() if line.strip()]
