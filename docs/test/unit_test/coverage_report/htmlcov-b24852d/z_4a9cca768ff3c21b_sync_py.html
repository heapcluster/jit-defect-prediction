66 交付模型时为空）。

71 raise AppError(40001, "参数非法：model_name 不在已加载模型清单")

81def _unpredicted( 

82 hashes: list[str], models: set[str], limit: int | None = None 

83) -> tuple[list[str], bool]: 

84 """按输入顺序挑出缺「models 中任一模型」预测行的哈希；分批 IN，不整表载入。 

85 

86 `models` 为空集时无缺口（全部视为已覆盖）。`limit` 非空时最多返回 limit 条， 

87 并在「还有更多」时把第二项置为 True。 

88 """ 

89 wanted = set(models) 

90 if not wanted: 90 ↛ 91line 90 didn't jump to line 91 because the condition on line 90 was never true

91 return [], False 

92 pending: list[str] = [] 

93 with SessionLocal() as session: 

94 for start in range(0, len(hashes), _PREDICTED_CHUNK): 

95 chunk = hashes[start : start + _PREDICTED_CHUNK] 

96 rows = session.execute( 

97 select(Prediction.commit_hash, Prediction.model_name).where( 

98 Prediction.commit_hash.in_(chunk) 

99 ) 

100 ) 

101 have: dict[str, set[str]] = {} 

102 for commit_hash, model_name in rows: 

103 have.setdefault(commit_hash, set()).add(model_name) 

104 covered = {h for h, names in have.items() if wanted <= names} 

105 for commit_hash in chunk: 

106 if commit_hash in covered: 

107 continue 

108 if limit is not None and len(pending) >= limit: 

109 return pending, True 

110 pending.append(commit_hash) 

111 return pending, False

124def _run_batch( 

125 *, 

126 mode: str, 

127 scope: str, 

128 models: set[str], 

129 plan: Callable[[str], tuple[int, list[str], bool | None]], 

130) -> dict: 

131 """取锁 → fetch → 按 plan 算待预测 → 逐条预测入库 → 归还锁。""" 

132 if not _sync_lock.acquire(blocking=False): 

133 raise AppError(50000, INTERNAL_ERROR_MESSAGE) 

134 sync_state.set_state( 

135 running=True, 

136 mode=mode, 

137 scope=scope, 

138 total=0, 

139 done=0, 

140 current_hash=None, 

141 started_at=datetime.now(UTC).replace(tzinfo=None), 

142 last_error=None, 

143 ) 

144 try: 

145 repo = config.SOURCE_REPO 

146 if repo is None or not (repo / ".git").exists(): 

147 raise AppError(40400, "sync source repository unavailable") 

148 

149 tip = fetch_latest_tip(str(repo)) 

150 scanned, pending, truncated = plan(str(repo)) 

151 sync_state.set_state(total=len(pending)) 

152 

153 model_names = sorted(models) 

154 extract_repo, release = ensure_extractable(str(repo), tip) 

155 try: 

156 results: list[dict] = [] 

157 skipped: list[dict] = [] 

158 for commit_hash in pending: 

159 sync_state.set_state(current_hash=commit_hash) 

160 try: 

161 predict_commit_all_models(commit_hash, repo=extract_repo, model_names=model_names) 

162 results.append({"commit_hash": commit_hash}) 

163 except AppError as exc: 

164 skipped.append( 

165 {"commit_hash": commit_hash, "code": exc.code, "message": exc.message} 

166 ) 

167 _update_done(results, skipped) 

168 finally: 

169 release() 

170 

171 logger.info( 

172 "sync done: mode=%s tip=%s scanned=%s pending=%s predicted=%s skipped=%s truncated=%s", 

173 mode, 

174 tip[:10], 

175 scanned, 

176 len(pending), 

177 len(results), 

178 len(skipped), 

179 truncated, 

180 ) 

181 data: dict = { 

182 "tip": tip, 

183 "scanned": scanned, 

184 "new": len(pending), 

185 "predicted": results, 

186 "skipped": skipped, 

187 } 

188 if truncated is not None: 

189 data["truncated"] = truncated 

190 return data 

191 except Exception as exc: 

192 sync_state.set_state(last_error=str(exc)[:200]) 

193 raise 

194 finally: 

195 sync_state.set_state(running=False) 

196 _sync_lock.release()

265@router.post("/api/sync") 

266def sync_endpoint( 

267 line: str | None = Query(default=None), 

268 model_name: str | None = Query(default=None), 

269 start_time: str | None = Query(default=None), 

270 end_time: str | None = Query(default=None), 

271) -> dict: 

272 """按架构 / 模型 / commit 时间区间**异步**补算入库（docs/api/sync.md）。 

273 

274 提交后立即返回；进度轮 `/api/sync/status`。同一时刻只允许一个同步任务， 

275 已在同步中时返回 50000（不排队）。 

276 """ 

277 if line is not None and line not in VALID_LINES: 

278 raise AppError(40001, "参数非法：line 取值需为 expert / semantic / fused / e2e") 

279 if model_name is not None and not model_registry.has(model_name): 

280 raise AppError(40001, "参数非法：model_name 不在已加载模型清单") 

281 start = parse_time_param(start_time, "start_time") if start_time else None 

282 end = parse_time_param(end_time, "end_time", "end") if end_time else None 

283 # 「检查 running + 置 running=True」必须在锁内原子完成：否则两个并发请求都能通过 

284 # 检查、各 spawn 一个 worker，败者 worker 的异常处理会把 running=False 写回， 

285 # 覆盖胜者正在跑的状态。 

286 with _sync_lock: 

287 if sync_state.get_sync_state()["running"]: 

288 raise AppError(50000, INTERNAL_ERROR_MESSAGE) 

289 # 在请求内先置 running=True：否则前端提交后立即轮询时，worker 可能还没启动， 

290 # 会读到上一次任务的 running=false 而误判「已完成」。 

291 sync_state.set_state( 

292 running=True, 

293 mode="manual", 

294 scope="latest", 

295 total=0, 

296 done=0, 

297 current_hash=None, 

298 started_at=datetime.now(UTC).replace(tzinfo=None), 

299 last_error=None, 

300 ) 

301 _spawn_sync_worker(lambda: sync_manual(line=line, model_name=model_name, start=start, end=end)) 

302 return ok({"accepted": True})
