20# 架构中文显示名：后端常量映射，不落库（docs/database/model.md） 

21LINE_LABELS = {"expert": "专家特征", "semantic": "语义", "fused": "融合", "e2e": "端到端"} 

22LINE_ORDER = ("expert", "semantic", "fused", "e2e") 

23 

24 

25@router.get("/api/models", response_model=schemas.ModelsResponse) 

26def list_models(line: str | None = Query(default=None)) -> dict: 

27 if line is not None and line not in LINE_ORDER: 

28 raise AppError(40001, "参数非法：line 取值需为 expert / semantic / fused / e2e") 

29 with SessionLocal() as session: 

30 stmt = select(Model.model_name, Model.line) 

31 if line is not None: 

32 stmt = stmt.where(Model.line == line) 

33 rows = session.execute(stmt).all() 

34 grouped: dict[str, list[str]] = {name: [] for name in LINE_ORDER} 

35 for model_name, model_line in rows: 

36 if model_line in grouped: 

37 grouped[model_line].append(model_name) 

38 lines = [ 

39 {"line": name, "label": LINE_LABELS[name], "models": sorted(grouped[name])} 

40 for name in LINE_ORDER 

41 if line is None or name == line 

42 ] 

43 return ok({"lines": lines})
