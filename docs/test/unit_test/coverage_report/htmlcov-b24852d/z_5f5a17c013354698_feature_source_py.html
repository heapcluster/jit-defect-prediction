29def extract(commit_hash: str, repo: Path | str | None = None) -> dict: 

30 """现算单条提交的 14 项特征。 

31 

32 `repo` 默认取 `config.SOURCE_REPO`；批量同步（/api/sync）在远端 tip 不是 

33 SOURCE_REPO 当前 HEAD 祖先时，传入专用临时工作树路径（str），这里统一转 Path， 

34 避免调用方传字符串时在 `source / ".git"` 处炸出 `str / str`（2026-10-26 修复）。 

35 """ 

36 source: Path | None = Path(repo) if repo is not None else config.SOURCE_REPO 

37 if source is None or not (source / ".git").exists(): 

38 raise RuntimeError("source repository unavailable") 

39 # 快查可达性：不可达提交直接拒，避免 extract_single_commit 先扫全量历史 

40 # （8 万+ 提交的 git log --numstat）再判「目标不是当前 HEAD 可达的非合并提交」。 

41 if not is_ancestor(str(source), commit_hash): 

42 raise ValueError("目标不是当前 HEAD 可达的非合并提交") 

43 return _extractor()(source, commit_hash)
