28def _with_database(url: str, database: str) -> str: 

29 """把显式 DSN 的库名换掉（保留口令与 query），pytest 切测试库用。""" 

30 from sqlalchemy.engine import make_url 

31 

32 try: 

33 return make_url(url).set(database=database).render_as_string(hide_password=False) 

34 except Exception: # noqa: BLE001 —— 非法 DSN 原样返回，把报错留给连接层 

35 return url

38def build_database_url(database: str | None = None) -> str: 

39 """按分字段拼 MySQL DSN；显式 `DATABASE_URL` 优先（CI / 容器一行搞定）。缺字段返回空串。 

40 

41 `database` 非空时覆盖库名：pytest 用它把 `DB_NAME` 换成 `DB_TEST_NAME`。 

42 """ 

43 explicit = os.getenv("DATABASE_URL", "").strip() 

44 if explicit: 

45 return _with_database(explicit, database) if database else explicit 

46 if not (DB_HOST and DB_USER): 

47 return "" 

48 credential = quote(DB_USER, safe="") 

49 if DB_PASSWORD: 

50 credential += f":{quote(DB_PASSWORD, safe='')}" 

51 return ( 

52 f"mysql+pymysql://{credential}@{DB_HOST}:{DB_PORT}/" 

53 f"{database or DB_NAME}?charset={DB_CHARSET}" 

54 )
