1"""时间工具：统一 UTC、ISO 8601、带 Z 后缀；以及查询时间参数的解析口径。 

2 

3契约三要求时间字段统一 UTC 且以 `_at` 结尾；`iso_z` 是唯一格式化入口。 

4`parse_time_param` 承载 §1「时间范围语义」（闭区间 + end 侧纯日期串含结束日全天）， 

5此前只在 `api/commits.py` 内部，`trends` / `quality` 需要同一口径，故上移到这里。 

6""" 

7 

8from __future__ import annotations 

9 

10import re 

11from datetime import UTC, datetime 

12 

13from app.errors import AppError 

14 

15 

16def iso_z(value: datetime | None) -> str | None: 

17 """`datetime` → `YYYY-MM-DDTHH:MM:SSZ`；`None` 原样返回（未预测 / 未标记字段）。""" 

18 if value is None: 

19 return None 

20 return value.replace(tzinfo=UTC).isoformat().replace("+00:00", "Z") 

21 

22 

23def now_naive_utc() -> datetime: 

24 """当前 UTC（naive）：MySQL DATETIME 是秒级存储，落库前统一去掉 tzinfo。""" 

25 return datetime.now(UTC).replace(tzinfo=None) 

26 

27 

28def parse_time_param(value: str, param: str, role: str = "start") -> datetime: 

29 """解析 ISO 8601 时间查询参数（契约三 §1 时间范围语义）。""" 

30 # 紧凑基本格式一律拒（含 Z/偏移后缀变体，#62 复审 1+3）：头 8 位数字后接 T 或 

31 # 结束即基本格式；fromisoformat 收但它绕过纯日期串判定，end 会静默停 00:00 

32 if re.fullmatch(r"\d{8}(T\S*)?", value): 

33 raise AppError( 

34 40001, f"参数非法：{param} 不接受紧凑基本格式，请用 YYYY-MM-DD 或带分隔符的 ISO 8601" 

35 ) 

36 try: 

37 parsed = datetime.fromisoformat(value.replace("Z", "+00:00")) 

38 except ValueError: 

39 raise AppError(40001, f"参数非法：{param} 需为 ISO 8601 时间") from None 

40 if parsed.tzinfo is not None: 

41 parsed = parsed.astimezone(UTC).replace(tzinfo=None) 

42 # 契约三 1.5 区间语义（#61）：闭区间；end 侧纯日期串含结束日全天， 

43 # start 侧纯日期串仍是当天 00:00:00（不能一刀切）；带时间分量按原样 

44 if role == "end" and re.fullmatch(r"\d{4}-\d{2}-\d{2}", value): 

45 parsed = parsed.replace(hour=23, minute=59, second=59, microsecond=999999) 

46 return parsed
