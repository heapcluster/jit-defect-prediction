27def resolve_model_name(model_name: str | None) -> str: 

28 """确定本次打分使用的模型名：显式给定时校验存在，否则取默认模型。 

29 

30 默认模型 = 已加载模型中 `version_key` 最大者（docs/database/model.md，`runtime_setting` 

31 与冠军模型热切换已下线），与列表 / 详情 / 趋势共用 `model_registry.default_model()`。 

32 

33 显式指定的模型区分两档错误（docs/todo「在线能力对齐」）：已在交付清单（`model` 表） 

34 但无在线权重 → 明确提示「仅离线评估」；完全不存在 → 「model_name 不存在」。 

35 """ 

36 if model_name: 

37 if model_registry.has(model_name): 

38 return model_name 

39 with SessionLocal() as session: 

40 delivered = session.execute( 

41 select(Model.model_name).where(Model.model_name == model_name) 

42 ).scalar_one_or_none() 

43 if delivered is not None: 

44 raise AppError(40001, "参数非法：该模型暂无在线权重，仅离线评估，无法在线预测") 

45 raise AppError(40001, "参数非法：model_name 不存在") 

46 entry = model_registry.default_model() 

47 if entry is None: 

48 raise AppError(50000, INTERNAL_ERROR_MESSAGE) 

49 return entry.name

53 """40400 文案按缺的是 commit 还是 feature 区分（契约三 message 逐字对齐）。"""

57def _extract_feature( 

58 commit_hash: str, repo=None, missing_message: str = "commit not found" 

59) -> dict: 

60 """现算单条提交的 14 项特征；抽取失败按异常类型映射为 40400/50000。 

61 

62 `missing_message` 由调用方传入：git 里也找不到该提交时，文案需与「库里缺哪一半」保持一致。 

63 """ 

64 try: 

65 return feature_source.extract(commit_hash, repo=repo) 

66 except ValueError as exc: 

67 raise AppError(40400, missing_message) from exc 

68 except Exception as exc: 

69 logger.exception("single-commit feature extraction failed") 

70 raise AppError(50000, INTERNAL_ERROR_MESSAGE) from exc

73def _upsert_commit_feature( 

74 session, commit_hash: str, extracted: dict 

75) -> tuple[Commit, CommitFeature]: 

76 """幂等写入 commit 与 commit_feature 行；并发双插时回落为查库。 

77 

78 返回 (commit, feature) 元组。 

79 """ 

80 committed_at = datetime.fromisoformat(extracted["committed_at"]) 

81 commit = session.execute( 

82 select(Commit).where(Commit.commit_hash == commit_hash) 

83 ).scalar_one_or_none() 

84 feature = session.execute( 

85 select(CommitFeature).where(CommitFeature.commit_hash == commit_hash) 

86 ).scalar_one_or_none() 

87 

88 if commit is None: 

89 commit = Commit( 

90 commit_hash=commit_hash, 

91 repo_name=extracted["repo_name"], 

92 author_name=extracted["author_name"], 

93 author_email=extracted["author_email"], 

94 committed_at=committed_at, 

95 message=extracted["message"], 

96 parent_hash=extracted["parent_hash"] or None, 

97 ) 

98 session.add(commit) 

99 if feature is None: 99 ↛ 107line 99 didn't jump to line 107 because the condition on line 99 was always true

100 feature = CommitFeature( 

101 commit_hash=commit_hash, 

102 feature_version=extracted["feature_version"], 

103 **extracted["features"], 

104 ) 

105 session.add(feature) 

106 

107 try: 

108 session.flush() 

109 except IntegrityError: 

110 session.rollback() 

111 commit = session.execute( 

112 select(Commit).where(Commit.commit_hash == commit_hash) 

113 ).scalar_one_or_none() 

114 feature = session.execute( 

115 select(CommitFeature).where(CommitFeature.commit_hash == commit_hash) 

116 ).scalar_one_or_none() 

117 if commit is None or feature is None: 

118 raise AppError(50000, INTERNAL_ERROR_MESSAGE) from None 

119 

120 return commit, feature

145def _upsert_predictions( 

146 session, commit_hash: str, feature: CommitFeature, model_names: list[str] | None = None 

147) -> list[dict]: 

148 """对指定模型逐模型打分并幂等写入 prediction 表（默认全部已加载模型）。 

149 

150 返回 [{model_name, risk_score}]。并发双插被联合主键挡住时回落为 update， 

151 保证幂等成功而非 500 —— 回落只重跑写入，不重复推理（分数已算好）。 

152 """ 

153 vector = [float(getattr(feature, name)) for name in config.FEATURE_COLUMNS] 

154 predicted_at = datetime.now(UTC).replace(tzinfo=None) 

155 entries = ( 

156 # 批量（sync）只对 expert 线打分：其它线的输入维度不同，需按线单独现算特征 

157 [entry for entry in model_registry.all_models() if entry.line == "expert"] 

158 if model_names is None 

159 else [model_registry.get_model(name) for name in model_names] 

160 ) 

161 scores = [(entry.name, model_registry.predict_proba(entry.model, vector)) for entry in entries] 

162 

163 try: 

164 _write_predictions(session, commit_hash, feature, scores, predicted_at) 

165 session.commit() 

166 except IntegrityError: 

167 session.rollback() 

168 _write_predictions(session, commit_hash, feature, scores, predicted_at) 

169 session.commit() 

170 

171 return [{"model_name": name, "risk_score": round(risk, 5)} for name, risk in scores]

174def predict_commit_all_models( 

175 commit_hash: str, repo=None, model_names: list[str] | None = None 

176) -> list[dict]: 

177 """对单条提交现算特征并对指定模型打分（默认全部已加载模型），结果写入库。 

178 

179 `commit_hash` 允许大写，内部归一化为小写后查库。 

180 `repo` 为 None 时取 `config.SOURCE_REPO`；批量同步可传入临时 worktree 路径。 

181 """ 

182 commit_hash = normalize_commit_hash(commit_hash) 

183 

184 with SessionLocal() as session: 

185 commit = session.execute( 

186 select(Commit).where(Commit.commit_hash == commit_hash) 

187 ).scalar_one_or_none() 

188 feature = session.execute( 

189 select(CommitFeature).where(CommitFeature.commit_hash == commit_hash) 

190 ).scalar_one_or_none() 

191 

192 if commit is None or feature is None: 192 ↛ 199line 192 didn't jump to line 199 because the condition on line 192 was always true

193 missing_message = _missing_message(commit) 

194 if config.SOURCE_REPO is None: 194 ↛ 195line 194 didn't jump to line 195 because the condition on line 194 was never true

195 raise AppError(40400, missing_message) 

196 extracted = _extract_feature(commit_hash, repo=repo, missing_message=missing_message) 

197 commit, feature = _upsert_commit_feature(session, commit_hash, extracted) 

198 

199 return _upsert_predictions(session, commit_hash, feature, model_names)

202def _vector_for_line(line: str, commit_hash: str, feature) -> list[float]: 

203 """按模型线的输入口径组装特征向量：expert 14 / semantic 768 / fused 782。 

204 

205 semantic/fused 用模型线预计算的语义 embedding（commit 不在表内 → 明确 40001，不编造）。 

206 """ 

207 expert = [float(getattr(feature, name)) for name in config.FEATURE_COLUMNS] 

208 if line == "expert": 

209 return expert 

210 embedding = semantic_source.embedding_for(commit_hash) 

211 if embedding is None: 

212 raise AppError(40001, f"参数非法：该提交暂无预计算语义向量，{line} 模型无法在线预测") 

213 if line == "semantic": 

214 return embedding.tolist() 

215 if line == "fused": 215 ↛ 217line 215 didn't jump to line 217 because the condition on line 215 was always true

216 return expert + embedding.tolist() 

217 raise AppError(40001, f"参数非法：{line} 模型暂不支持在线预测")

220def predict_single(commit_hash: str, model_name: str, repo=None) -> dict: 

221 """对单条提交用指定模型打分，返回契约三 1.5 第 4 节的完整响应体。 

222 

223 错误码：`model_name` 不在已加载清单 → 40001；提交/特征缺失且 

224 `SOURCE_REPO` 未配置 → 40400；注册表整体为空 → 50000。 

225 """ 

226 commit_hash = normalize_commit_hash(commit_hash) 

227 

228 with SessionLocal() as session: 

229 commit = session.execute( 

230 select(Commit).where(Commit.commit_hash == commit_hash) 

231 ).scalar_one_or_none() 

232 feature = session.execute( 

233 select(CommitFeature).where(CommitFeature.commit_hash == commit_hash) 

234 ).scalar_one_or_none() 

235 

236 if commit is None or feature is None: 

237 missing_message = _missing_message(commit) 

238 if config.SOURCE_REPO is None: 238 ↛ 239line 238 didn't jump to line 239 because the condition on line 238 was never true

239 raise AppError(40400, missing_message) 

240 extracted = _extract_feature(commit_hash, repo=repo, missing_message=missing_message) 

241 commit, feature = _upsert_commit_feature(session, commit_hash, extracted) 

242 

243 model_name = resolve_model_name(model_name) 

244 line = model_registry.line_of(model_name) 

245 if line == "e2e": 245 ↛ 247line 245 didn't jump to line 247 because the condition on line 245 was never true

246 # e2e 是完整 CodeBERT 分类头：读提交文本经模型前向，无 pkl/向量组装 

247 risk = e2e_source.predict_proba(commit_hash) 

248 if risk is None: 

249 raise AppError(40001, "参数非法：e2e 模型所需文本或编码器不可用，无法在线预测") 

250 explanation: list[dict] = [] 

251 entry_name = model_name 

252 else: 

253 try: 

254 entry = model_registry.get_model(model_name) 

255 except model_registry.ModelUnavailable as exc: 

256 raise AppError(40001, f"参数非法：{exc}") from exc 

257 vector = _vector_for_line(line, commit_hash, feature) 

258 risk = model_registry.predict_proba(entry.model, vector) 

259 # SHAP 只对 expert 线算（768/782 维重模型归因慢且无必要）；其余返回空解释 

260 explanation = ( 

261 with_labels(model_registry.explain(entry, vector)) if line == "expert" else [] 

262 ) 

263 entry_name = entry.name 

264 predicted_at = datetime.now(UTC).replace(tzinfo=None) 

265 

266 row = session.execute( 

267 select(Prediction).where( 

268 Prediction.commit_hash == commit_hash, 

269 Prediction.model_name == model_name, 

270 ) 

271 ).scalar_one_or_none() 

272 if row is None: 

273 row = Prediction(commit_hash=commit_hash, model_name=model_name) 

274 session.add(row) 

275 row.risk_score = risk 

276 row.predicted_at = predicted_at 

277 session.commit() 

278 

279 # 读回落库值再进响应：MySQL DATETIME 是秒级精度，in-memory 里的微秒会被截断， 

280 # 若直接返回本地值，「重复预测」时响应与库内行会对不上账（集成测试实测过）。 

281 stored_predicted_at = row.predicted_at 

282 

283 features = { 

284 name: (int(getattr(feature, name)) if name == "fix" else float(getattr(feature, name))) 

285 for name in config.FEATURE_COLUMNS 

286 } 

287 

288 return { 

289 "commit_hash": commit_hash, 

290 "model_name": entry_name, 

291 "risk_score": round(risk, 5), 

292 "predicted_at": iso_z(stored_predicted_at), 

293 "features": features, 

294 "explanation": explanation, 

295 }
